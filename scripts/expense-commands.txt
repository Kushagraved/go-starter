# Sample command file for the expense-sharing CLI.
# Run with:
#   make infra && make db-schema
#   go run ./cmd/script -config-path config/dev/config.yaml -file scripts/expense-commands.txt
#
# Blank lines and lines starting with '#' are ignored. Add_User prints the new
# user id assigned by the database; note them and use them below.

Add_User Alice alice@example.com 9876543210
Add_User Bob bob@example.com 9876543211
Add_User Carol carol@example.com 9876543212

# Equal: Alice (id 1) pays 100, split among 1,2,3.
Add_Expense Dinner 100 1 1 3 Equal 1 2 3

# Exact: Bob (id 2) pays 40, Alice owes 25, Bob owes 15.
Add_Expense Cab 40 2 2 2 Exact 1 2 25 15

# Percent: Carol (id 3) pays 300, split 50/25/25.
Add_Expense Trip 300 3 3 3 Percent 1 2 3 50 25 25

Show_All_User_Balance
Show_User_Data 1
Show_User_Expense 1
